<script>
const board=document.getElementById("board");
const turnText=document.getElementById("turn");
const diceText=document.getElementById("dice");
const message=document.getElementById("message");

const players=[
  {name:"BLACK",cls:"black",symbol:"🦅"},
  {name:"WHITE",cls:"white",symbol:"🐎"},
  {name:"GREY",cls:"grey",symbol:"🦅"},
  {name:"PINK",cls:"pink",symbol:"🐎"}
];

let currentPlayer=0;
let dice=0;
let rolled=false;

/* Simple circular path */
const path=[];
for(let i=0;i<52;i++){
  path.push(i);
}

/* Token positions */
let tokens=players.map(()=>[0,0,0,0]);

function createBoard(){
  board.innerHTML="";

  for(let r=0;r<15;r++){
    for(let c=0;c<15;c++){
      const cell=document.createElement("div");
      cell.className="cell";

      /* Four home areas */
      if(r<6 && c<6) cell.classList.add("home-black");
      else if(r<6 && c>8) cell.classList.add("home-white");
      else if(r>8 && c<6) cell.classList.add("home-grey");
      else if(r>8 && c>8) cell.classList.add("home-pink");
      else cell.classList.add("path");

      if(r>=6 && r<=8 && c>=6 && c<=8){
        cell.className="cell center";
      }

      board.appendChild(cell);
    }
  }

  drawTokens();
}

function getPathCell(index){
  /*
    Board ke outer route ke approximate cells.
  */
  const route=[];

  for(let c=0;c<15;c++) route.push([6,c]);
  for(let r=7;r<15;r++) route.push([r,14]);
  for(let c=13;c>=0;c--) route.push([14,c]);
  for(let r=13;r>=0;r--) route.push([r,0]);
  for(let c=1;c<15;c++) route.push([0,c]);
  for(let r=1;r<6;r++) route.push([r,14]);

  return route[index % route.length];
}

function drawTokens(){
  const cells=[...document.querySelectorAll(".cell")];

  players.forEach((p,pi)=>{
    tokens[pi].forEach((pos,ti)=>{
      let rc;

      if(pos===0){
        const homes=[
          [[1,1],[1,3],[3,1],[3,3]],
          [[1,11],[1,13],[3,11],[3,13]],
          [[11,1],[11,3],[13,1],[13,3]],
          [[11,11],[11,13],[13,11],[13,13]]
        ];
        rc=homes[pi][ti];
      }else{
        rc=getPathCell((pos-1)%52);
      }

      const index=rc[0]*15+rc[1];
      const token=document.createElement("div");

      token.className="token "+p.cls;
      token.innerHTML=p.symbol;

      token.onclick=()=>{
        if(pi!==currentPlayer || !rolled){
          return;
        }

        if(dice===6 && pos===0){
          tokens[pi][ti]=1;
        }else if(pos>0){
          tokens[pi][ti]+=dice;
          if(tokens[pi][ti]>52) tokens[pi][ti]=52;
        }else{
          message.innerText="Goti nikalne ke liye 6 chahiye.";
          return;
        }

        rolled=false;
        diceText.innerText="🎲";
        nextTurn();
      };

      cells[index].appendChild(token);
    });
  });
}

function rollDice(){
  if(rolled){
    message.innerText="Pehle apni goti select karein.";
    return;
  }

  dice=Math.floor(Math.random()*6)+1;
  diceText.innerText=["","⚀","⚁","⚂","⚃","⚄","⚅"][dice];
  rolled=true;

  message.innerText=
    dice===6
    ? "6 aaya! Goti par click karein."
    : "Apni goti par click karein.";
}

function nextTurn(){
  if(dice!==6){
    currentPlayer=(currentPlayer+1)%players.length;
  }

  turnText.innerText="Turn: "+players[currentPlayer].name;
  message.innerText=players[currentPlayer].name+" ki baari.";
  createBoard();
}

createBoard();
</script>
